## Installation

<Steps>
    <Step>
        ### Install the plugin
        
        ```package-install
        @node-zugferd/instrumentation @opentelemetry/api
        ```
    </Step>
    <Step>
        ### Add the plugin to your instance
        
        ```ts title="invoicer.ts"
        import { zugferd } from "node-zugferd";
        import { en16931 } from "@node-zugferd/en-16931";
        import { instrumentation } from "@node-zugferd/instrumentation"; // [!code highlight]
        
        export const invoicer = zugferd({
          profiles: [en16931],
          plugins: [
            instrumentation(), // [!code highlight]
          ],
          // ...
        });
        ```
    </Step>
</Steps>
